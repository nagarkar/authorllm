% folio centred in the foot. \AuthorLMRunningBook is set by the
% exporter (header-includes) to the manuscript title.

\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE]{\small\itshape\AuthorLMRunningBook}
\fancyhead[RO]{\small\itshape\nouppercase{\rightmark}}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\small\thepage}}
