% The review filter invokes this only for the default PDF profile. A
% --print-ready export never emits the invocation, leaving a clean PDF.
% #3 is the recipient line from export --notice (empty: standard footer).
\providecommand{\AuthorLMReviewMode}[3]{%
  \AuthorLMEnableReviewPageStyle{}
  \AddToShipoutPictureBG{%
    \AtPageCenter{%
      \makebox[0pt][c]{%
        \rotatebox{45}{%
          \parbox{0.72\paperwidth}{%
            \centering\fontsize{64}{66}\selectfont\color[gray]{0.92}%
            Confidential\\Review%
          }%
        }%
      }%
    }%
  }%
  \AddToShipoutPictureFG{%
    \AtPageLowerLeft{%
      \raisebox{0.3in}{%
        \makebox[\paperwidth][c]{%
          \parbox{0.82\paperwidth}{\centering\scriptsize
            \if\relax\detokenize{#3}\relax\else
              \textbf{#3}\\[1pt]%
            \fi
            CONFIDENTIAL REVIEW DRAFT · DO NOT COPY, SHARE, OR DISTRIBUTE ·
            \textcopyright{} #2 #1. All rights reserved.\hspace{1.5em}\thepage}%
        }%
      }%
    }%
  }%
}
